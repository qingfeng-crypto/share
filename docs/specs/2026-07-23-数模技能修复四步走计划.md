### P1.2 补齐 check_kb_layout.py（高严重度）

**文件**: `references/roles/编程手核心/scripts/check_kb_layout.py`（**新建**）

**当前问题**: SKILL.md "知识库增改铁律" 第 4 步（主 SKILL.md L228-231）硬性要求运行此脚本，但文件不存在。

**依赖**: 无 | **并行**: 独立 | **预估**: 15 分钟
**风险**: 低（首次创建，不影响现有功能）
**回滚**: 删除该文件即可

**完整脚本**：

```python
#!/usr/bin/env python3
"""检查知识库目录结构是否符合"章/节/单md"三级规范。

使用方法: python check_kb_layout.py
返回值: 0 = 结构合规, 1 = 发现结构问题

技能根目录解析: 从本脚本路径向上回溯 4 级（scripts/ → 编程手核心/ → roles/ → references/ → skill 根）
"""

import sys
from pathlib import Path

# ── 路径解析 ──────────────────────────────────────────
SCRIPT_DIR = Path(__file__).resolve().parent
# 从 scripts/ → 编程手核心/ → roles/ → references/ → skill 根
SKILL_BASE = SCRIPT_DIR.parents[3]
TARGET_DIR = SKILL_BASE / "数学建模比赛完整学习知识"
CHAPTER_GLOB = "第*章*"


def check_structure() -> list[str]:
    """遍历目标目录，检查三级结构合规性。返回问题列表，空 = 合规。"""
    issues: list[str] = []

    if not TARGET_DIR.exists():
        issues.append(f"[SKIP] 目标目录不存在（跳过结构检查）: {TARGET_DIR}")
        return issues

    chapters = sorted([d for d in TARGET_DIR.iterdir() if d.is_dir()])

    for ch_dir in chapters:
        # 检查章目录下是否有平铺 .md（直接放在章下而非节下）
        top_mds = [
            f for f in ch_dir.iterdir()
            if f.is_file() and f.suffix == ".md"
        ]
        for f in top_mds:
            # Obsidian 系统文件豁免
            if f.name in ("_index.md", "_template.md"):
                continue
            rel = f.relative_to(TARGET_DIR)
            issues.append(f"[WARN] 平铺 md（应放在节目录下）: {rel}")

        # 检查节目录
        sections = sorted([d for d in ch_dir.iterdir() if d.is_dir()])
        for sec_dir in sections:
            sec_mds = [
                f for f in sec_dir.iterdir()
                if f.is_file() and f.suffix == ".md"
            ]
            if not sec_mds:
                rel = sec_dir.relative_to(TARGET_DIR)
                issues.append(f"[WARN] 空节目录（无 .md 文件）: {rel}")

    return issues


def main() -> int:
    issues = check_structure()

    if not issues:
        print("[OK] 所有章节均符合"章/节/单md"三级结构")
        return 0

    if issues and issues[0].startswith("[SKIP]"):
        print(issues[0])
        return 0

    print(f"发现 {len(issues)} 个结构问题:")
    for item in issues:
        print(f"  {item}")
    return 1


if __name__ == "__main__":
    sys.exit(main())
```

```python
import sys
from pypdf import PdfReader


def has_fillable_fields(pdf_path: str) -> bool:
    """检查 PDF 是否包含可填表单字段。"""
    reader = PdfReader(pdf_path)
    return reader.get_fields() is not None


def main() -> int:
    if len(sys.argv) < 2:
        print("Usage: python check_fillable_fields.py <pdf_path>")
        return 1
    try:
        path = sys.argv[1]
        if has_fillable_fields(path):
            print("This PDF has fillable form fields")
        else:
            print("This PDF does not have fillable form fields; you will need to visually determine where to enter data")
        return 0
    except Exception as e:
        print(f"Error: {e}")
        return 1


if __name__ == "__main__":
    sys.exit(main())
```
